// 1. 회원가입 및 로그인 처리 async function handleAuth() { const email = document.getElementById('authEmail').value.trim(); const password = document.getElementById('authPassword').value.trim(); const nickname = document.getElementById('authNickname').value.trim(); if (!email || !password) return alert('이메일과 비밀번호를 입력해주세요.'); const authBtn = document.getElementById('authMainBtn'); authBtn.innerText = '처리 중...'; authBtn.disabled = true; try { if (isSignUpMode) { // 이메일 인증 활성화 상태의 가입 요청 (Redirect URL 명시) const { data, error } = await supabaseClient.auth.signUp({ email, password, options: { emailRedirectTo: window.location.origin + window.location.pathname } }); if (error) throw error; if (data.user) { // 프로필 기본 정보 생성 await supabaseClient.from('profiles').upsert([{ id: data.user.id, email: email, nickname: nickname || '사용자' }]); // 세션이 없다면 = 이메일 인증이 필요한 상태 if (!data.session) { alert(`[${email}] 주소로 인증 메일이 발송되었습니다.\n메일함의 링크를 클릭한 후 로그인해 주세요.`); toggleAuthMode(); // 로그인 화면으로 자동 전환 return; } await checkSession(); } } else { // 일반 로그인 const { data, error } = await supabaseClient.auth.signInWithPassword({ email, password }); if (error) { if (error.message.includes('Email not confirmed')) { throw new Error('이메일 인증이 완료되지 않았습니다. 메일함에서 인증 링크를 클릭해 주세요.'); } throw error; } await checkSession(); } } catch (err) { alert((isSignUpMode ? '회원가입' : '로그인') + ' 실패: ' + err.message); } finally { authBtn.innerText = isSignUpMode ? '회원가입' : '로그인'; authBtn.disabled = false; } } // 2. 인증 완료 후 메일 링크를 타고 앱에 돌아왔을 때 감지 supabaseClient.auth.onAuthStateChange(async (event, session) => { if (event === 'SIGNED_IN' || event === 'USER_UPDATED') { await checkSession(); } });
